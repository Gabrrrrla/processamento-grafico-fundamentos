cmake_minimum_required(VERSION 3.11)
project(TrabalhoGA)

# Define o padrão do C++
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED True)

# Garante que as bibliotecas baixadas sejam compiladas como estáticas
set(BUILD_SHARED_LIBS OFF)

# Ativa o FetchContent
include(FetchContent)

# --- Dependências baixadas via FetchContent ---
FetchContent_Declare(
  glfw
  GIT_REPOSITORY https://github.com/glfw/glfw.git
  GIT_TAG "3.4"
)
FetchContent_Declare(
  glm
  GIT_REPOSITORY https://github.com/g-truc/glm.git
  GIT_TAG "1.0.1"
)
FetchContent_Declare(
  stb_image
  GIT_REPOSITORY https://github.com/nothings/stb.git
  GIT_TAG master
)
FetchContent_Declare(
  soil2
  GIT_REPOSITORY https://github.com/SpartanJ/SOIL2.git
  GIT_TAG master
)

# Torna as dependências disponíveis para o projeto
FetchContent_MakeAvailable(glfw glm stb_image soil2)

# --- Configuração do GLAD (local) ---
# Assumindo que glad.c está na pasta 'libs'
set(GLAD_C_FILE "${CMAKE_SOURCE_DIR}/libs/glad.c")
# Assumindo que glad.h está em 'libs/glad/glad.h'
include_directories(${CMAKE_SOURCE_DIR}/libs)

# --- Configuração do Executável ---
if(WIN32)
    set(OPENGL_LIBS opengl32)
else()
    find_package(OpenGL REQUIRED)
    set(OPENGL_LIBS ${OPENGL_gl_LIBRARY})
endif()

get_filename_component(EXE_NAME main.cpp NAME_WE)
file(GLOB EXE_SOURCES main.cpp Classes/*.cpp)

# Adiciona glad.c aos fontes do executável
add_executable(${EXE_NAME} ${EXE_SOURCES} ${GLAD_C_FILE})

# Configura os diretórios de include
target_include_directories(${EXE_NAME} PRIVATE
    ${glm_SOURCE_DIR}
    ${stb_image_SOURCE_DIR}
    ${soil2_SOURCE_DIR}/src/SOIL2
)

# --- Linkagem Final ---
target_link_libraries(${EXE_NAME} PRIVATE
    glfw
    ${OPENGL_LIBS}
    glm::glm
    soil2
)
